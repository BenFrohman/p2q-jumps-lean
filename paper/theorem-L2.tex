\begin{theorem}[Theorem~L2: strengthened Lean-certified statement, v1.4]
\label{thm:L2}
Let \(p\) and \(q\) be distinct primes, \(n=p^2q\) and \(N=n+12\), and suppose \(\ph(N)=\ph(n)+12\) and \(\sig(N)=\sig(n)+12\). Then:
\begin{enumerate}
\item[\textup{(i)}] \(N\) is not prime, and \(N\) is not a product of two distinct primes;
\item[\textup{(ii)}] \(N\) is not a product of three distinct primes;
\item[\textup{(iii)}] \(N\) is not of the form \(r^2st\) with pairwise distinct primes \(r,s,t\ge 5\);
\item[\textup{(iv)}] \(N\) is not of the form \(r^3s\) with distinct primes \(r,s\);
\item[\textup{(v)}] \(n>10^4\);
\item[\textup{(vi)}] \(N\) is not of the form \(r^k\) with \(r\) prime and \(k\ge 1\), and \(N\) is not of the form \(r^as\) with distinct primes \(r,s\) and \(a\ge 2\);
\item[\textup{(vii)}] if \(\omega(N)\ge 4\), then \(q\ge 3p^2\) and every prime \(r\) dividing \(N\) satisfies \(r>p\).
\end{enumerate}
\end{theorem}
